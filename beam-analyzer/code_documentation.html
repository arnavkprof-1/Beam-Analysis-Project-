  89    /**
  90     * Static Equilibrium Analysis
  91     * Computes reaction forces and reaction moments.
  92     */
  93    calculateReactions() {
  94      const L = this.length;
  95      let totalDownwardForce = 0;
  96      let sumMomentsAboutOrigin = 0; // Clockwise positive moments about x=0
  97      let steps = [];
  98  
  99      // Sum from point loads (downward load F at x produces clockwise moment about x=0: +F*x)
 100      for (const pl of this.pointLoads) {
 101        totalDownwardForce += pl.magnitude;
 102        sumMomentsAboutOrigin += pl.magnitude * pl.x;
 103      }
 104  
 105      // Sum from UDL loads
 106      for (const udl of this.udlLoads) {
 107        const span = Math.max(0, udl.x2 - udl.x1);
 108        const totalUDLForce = udl.magnitude * span;
 109        const centroid = (udl.x1 + udl.x2) / 2;
 110        totalDownwardForce += totalUDLForce;
 111        sumMomentsAboutOrigin += totalUDLForce * centroid;
 112      }
 113  
 114      // Concentrated applied moments (clockwise positive adds directly)
 115      for (const m of this.moments) {
 116        sumMomentsAboutOrigin += m.magnitude;
 117      }
 118  
 119      let reactions = {
 120        R1: 0, // at support A
 121        R2: 0, // at support B
 122        M_fixed: 0, // fixed clamp reaction moment if cantilever
 123        posA: this.supportA.x,
 124        posB: this.supportB.x,
 125        isStable: true,
 126        statusMessage: 'Equilibrium Satisfied',
 127        totalDownwardForce,
 128        totalUpwardReaction: 0,
 129        balanceError: 0,
 130        steps: []
 131      };
 132  
 133      if (this.supportType === 'cantilever-left') {
 134        // Clamped at x = 0
 135        // Upward reaction force R1 = total downward force
 136        // Counterclockwise clamp moment M_fixed balances sum of clockwise moments
 137        const R1 = totalDownwardForce;
 138        const M_fixed = -sumMomentsAboutOrigin; // Counterclockwise reaction moment
 139        reactions.R1 = R1;
 140        reactions.R2 = 0;
 141        reactions.M_fixed = M_fixed;
 142        reactions.posA = 0;
 143        reactions.totalUpwardReaction = R1;
 144        reactions.balanceError = Math.abs(R1 - totalDownwardForce);
 145  
 146        reactions.steps = [
 147          `Cantilever fixed at x = 0.00 m:`,
 148          `∑ Fy = 0  =>  R_fixed = Total Downward Load = ${R1.toFixed(3)} kN ↑`,
 149          `∑ M(x=0) = 0  =>  M_reaction = ${Math.abs(M_fixed).toFixed(3)} kN·m (${M_fixed <= 0 ? 'Counter-Clockwise' : 'Clockwise'})`
 150        ];
 151  
 152        return reactions;
 153      }
 154  
 155      if (this.supportType === 'cantilever-right') {
 156        // Clamped at x = L
 157        const R2 = totalDownwardForce;
 158        // Moments about x = L:
 159        // A downward load at x produces counterclockwise moment about x=L: -F*(L - x)
 160        let sumMomentsAboutL = 0;
 161        for (const pl of this.pointLoads) {
 162          sumMomentsAboutL -= pl.magnitude * (L - pl.x);
 163        }
 164        for (const udl of this.udlLoads) {
 165          const span = Math.max(0, udl.x2 - udl.x1);
 166          const force = udl.magnitude * span;
 167          const centroid = (udl.x1 + udl.x2) / 2;
 168          sumMomentsAboutL -= force * (L - centroid);
 169        }
 170        for (const m of this.moments) {
 171          sumMomentsAboutL += m.magnitude;
 172        }
 173  
 174        const M_fixed = -sumMomentsAboutL;
 175        reactions.R1 = 0;
 176        reactions.R2 = R2;
 177        reactions.M_fixed = M_fixed;
 178        reactions.posB = L;
 179        reactions.totalUpwardReaction = R2;
 180        reactions.balanceError = Math.abs(R2 - totalDownwardForce);
 181  
 182        reactions.steps = [
 183          `Cantilever fixed at x = ${L.toFixed(2)} m:`,
 184          `∑ Fy = 0  =>  R_fixed = Total Downward Load = ${R2.toFixed(3)} kN ↑`,
 185          `∑ M(x=L) = 0  =>  M_reaction = ${Math.abs(M_fixed).toFixed(3)} kN·m`
 186        ];
 187  
 188        return reactions;
 189      }
 190  
 191      // Two-support system (Simply supported or Overhanging)
 192      const xA = this.supportA.x;
 193      const xB = this.supportB.x;
 194      const spanAB = xB - xA;
 195  
 196      if (Math.abs(spanAB) < 1e-4) {
 197        reactions.isStable = false;
 198        reactions.statusMessage = 'Unstable: Supports A and B coincide (Span = 0). Cannot resist overturning.';
 199        return reactions;
 200      }
 201  
 202      // Moment equilibrium about Support A (taking clockwise as positive):
 203      // R2 acts upward at xB, producing counterclockwise moment about xA: -R2 * (xB - xA)
 204      // Applied loads produce moment about xA:
 205      // Downward force at x: moment about xA = +F * (x - xA)
 206      let sumMomentsAboutA = 0;
 207  
 208      for (const pl of this.pointLoads) {
 209        sumMomentsAboutA += pl.magnitude * (pl.x - xA);
 210      }
 211  
 212      for (const udl of this.udlLoads) {
 213        const span = Math.max(0, udl.x2 - udl.x1);
 214        const force = udl.magnitude * span;
 215        const centroid = (udl.x1 + udl.x2) / 2;
 216        sumMomentsAboutA += force * (centroid - xA);
 217      }
 218  
 219      for (const m of this.moments) {
 220        sumMomentsAboutA += m.magnitude;
 221      }
 222  
 223      // ∑ M_A = 0 => sumMomentsAboutA - R2 * (xB - xA) = 0
 224      // => R2 = sumMomentsAboutA / (xB - xA)
 225      const R2 = sumMomentsAboutA / spanAB;
 226      const R1 = totalDownwardForce - R2;
 227  
 228      reactions.R1 = R1;
 229      reactions.R2 = R2;
 230      reactions.totalUpwardReaction = R1 + R2;
 231      reactions.balanceError = Math.abs((R1 + R2) - totalDownwardForce);
 232  
 233      reactions.steps = [
 234        `1. Equilibrium of Moments about Support A (x = ${xA.toFixed(2)} m):`,
 235        `   ∑ M_A = 0  =>  R_B · (${xB.toFixed(2)} - ${xA.toFixed(2)}) - (${sumMomentsAboutA.toFixed(3)} kN·m) = 0`,
 236        `   R_B = ${sumMomentsAboutA.toFixed(3)} / ${spanAB.toFixed(3)} = ${R2.toFixed(3)} kN ${R2 >= 0 ? '↑' : '↓'}`,
 237        `2. Vertical Force Equilibrium:`,
 238        `   ∑ F_y = 0  =>  R_A + R_B = Total Applied Load (${totalDownwardForce.toFixed(3)} kN)`,
 239        `   R_A = ${totalDownwardForce.toFixed(3)} - (${R2.toFixed(3)}) = ${R1.toFixed(3)} kN ${R1 >= 0 ? '↑' : '↓'}`,
 240        `3. Equilibrium Verification:`,
 241        `   |∑ F_y| = ${reactions.balanceError.toExponential(2)} kN ≈ 0.00 kN [PASS]`
 242      ];
 243  
 244      return reactions;
 245    }

 247    /**
 248     * Internal Section Mechanics: Evaluates V(x) and M(x) at any section x
 249     * Using Euler-Bernoulli left-to-right section cut:
 250     * Upward forces to left cause positive shear (+V)
 251     * Downward loads to left cause negative shear (-V)
 252     * Sagging moment (compression top, tension bottom) is positive:
 253     * - Upward forces to left cause positive moment: +R * (x - x_R)
 254     * - Downward loads to left cause negative moment: -P * (x - x_P)
 255     * - Clockwise applied moment to left causes positive bending moment
 256     */
 257    evaluateAt(x, reactions) {
 258      let V = 0;
 259      let M = 0;
 260  
 261      // 1. Reactions to the left of x
 262      if (this.supportType === 'cantilever-left') {
 263        if (x > 0) {
 264          V += reactions.R1;
 265          M += reactions.R1 * x + reactions.M_fixed;
 266        }
 267      } else if (this.supportType === 'cantilever-right') {
 268        // Right cantilever: reaction is at x=L, so no reaction to left if x < L
 269        // Only applied loads to the left act
 270      } else {
 271        // Two supports
 272        if (x > reactions.posA) {
 273          V += reactions.R1;
 274          M += reactions.R1 * (x - reactions.posA);
 275        }
 276        if (x > reactions.posB) {
 277          V += reactions.R2;
 278          M += reactions.R2 * (x - reactions.posB);
 279        }
 280      }
 281  
 282      // 2. Point loads to the left of x
 283      for (const pl of this.pointLoads) {
 284        if (x > pl.x) {
 285          V -= pl.magnitude;
 286          M -= pl.magnitude * (x - pl.x);
 287        }
 288      }
 289  
 290      // 3. UDL loads to the left of x
 291      for (const udl of this.udlLoads) {
 292        if (x > udl.x1) {
 293          const activeSpan = Math.min(x, udl.x2) - udl.x1;
 294          if (activeSpan > 0) {
 295            const force = udl.magnitude * activeSpan;
 296            const centroid = udl.x1 + activeSpan / 2;
 297            V -= force;
 298            M -= force * (x - centroid);
 299          }
 300        }
 301      }
 302  
 303      // 4. Concentrated moments to the left of x
 304      for (const m of this.moments) {
 305        if (x > m.x) {
 306          M += m.magnitude;
 307        }
 308      }
 309  
 310      // Flip moment sign if civil/hogging-positive convention is selected
 311      const signFactor = this.signConvention === 'hogging-positive' ? -1 : 1;
 312  
 313      return {
 314        x,
 315        V,
 316        M: M * signFactor
 317      };
 318    }

 321     * Continuous Analysis over 2000 points with exact step discontinuity capture
 322     */
 323    solve() {
 324      const reactions = this.calculateReactions();
 325      const L = this.length;
 326      const numPoints = 2000;
 327  
 328      // Collect all critical discontinuity boundaries
 329      const criticalX = new Set([0, L]);
 330      if (this.supportType === 'cantilever-left') {
 331        criticalX.add(0);
 332      } else if (this.supportType === 'cantilever-right') {
 333        criticalX.add(L);
 334      } else {
 335        criticalX.add(this.supportA.x);
 336        criticalX.add(this.supportB.x);
 337      }
 338  
 339      for (const pl of this.pointLoads) criticalX.add(pl.x);
 340      for (const udl of this.udlLoads) {
 341        criticalX.add(udl.x1);
 342        criticalX.add(udl.x2);
 343      }
 344      for (const m of this.moments) criticalX.add(m.x);
 345  
 346      const sortedCrit = Array.from(criticalX).sort((a, b) => a - b);
 347  
 348      // Build sampling x-points list including exact dual points (x - eps, x + eps)
 349      const eps = 1e-6;
 350      const xSamplePoints = [];
 351  
 352      // Uniform grid
 353      const dx = L / (numPoints - 1);
 354      for (let i = 0; i < numPoints; i++) {
 355        xSamplePoints.push(i * dx);
 356      }
 357  
 358      // Inject discontinuity pairs
 359      for (const cx of sortedCrit) {
 360        if (cx > 0 && cx < L) {
 361          xSamplePoints.push(Math.max(0, cx - eps));
 362          xSamplePoints.push(Math.min(L, cx + eps));
 363        } else if (cx === 0) {
 364          xSamplePoints.push(eps);
 365        } else if (cx === L) {
 366          xSamplePoints.push(L - eps);
 367        }
 368      }
 369  
 370      // Sort and remove duplicates within 1e-9
 371      xSamplePoints.sort((a, b) => a - b);
 372      const uniqueX = [];
 373      for (let i = 0; i < xSamplePoints.length; i++) {
 374        if (i === 0 || Math.abs(xSamplePoints[i] - xSamplePoints[i - 1]) > 1e-7) {
 375          uniqueX.push(xSamplePoints[i]);
 376        }
 377      }
 378  
 379      // Compute V(x) and M(x) across points
 380      const points = [];
 381      let maxV = -Infinity;
 382      let minV = Infinity;
 383      let maxAbsV = 0;
 384      let maxM = -Infinity;
 385      let minM = Infinity;
 386      let maxAbsM = 0;
 387      let maxVPos = 0;
 388      let maxMPos = 0;
 389      let minMPos = 0;
 390  
 391      for (const x of uniqueX) {
 392        const res = this.evaluateAt(x, reactions);
 393        points.push(res);
 394  
 395        if (res.V > maxV) maxV = res.V;
 396        if (res.V < minV) minV = res.V;
 397        if (Math.abs(res.V) > maxAbsV) {
 398          maxAbsV = Math.abs(res.V);
 399          maxVPos = res.x;
 400        }
 401  
 402        if (res.M > maxM) {
 403          maxM = res.M;
 404          maxMPos = res.x;
 405        }
 406        if (res.M < minM) {
 407          minM = res.M;
 408          minMPos = res.x;
 409        }
 410        if (Math.abs(res.M) > maxAbsM) {
 411          maxAbsM = Math.abs(res.M);
 412        }
 413      }
 414  
 415      // Detect Points of Zero Shear (where V(x) crosses zero or jumps across zero)
 416      const zeroShearPoints = this.findZeroShearPoints(points, sortedCrit, reactions);
 417  
 418      // Detect Points of Contraflexure / Inflection (where M(x) = 0)
 419      const contraflexurePoints = this.findContraflexurePoints(points, sortedCrit, reactions);
 420  
 421      // Compute maximum bending stress σ = |M_max| / Z
 422      // M in kN*m = 10^6 N*mm
 423      // Z in cm^3 = 10^3 mm^3
 424      // σ = (M * 10^6 N*mm) / (Z * 10^3 mm^3) = (M / Z) * 1000 MPa (N/mm^2)
 425      const peakMomentAbs = Math.max(Math.abs(maxM), Math.abs(minM));
 426      const peakStress = this.sectionModulus > 0 ? (peakMomentAbs * 1000) / this.sectionModulus : 0; // MPa
 427      const stressRatio = this.allowableStress > 0 ? (peakStress / this.allowableStress) * 100 : 0; // %
 428      const safetyFactor = peakStress > 0 ? this.allowableStress / peakStress : Infinity;
 429  
 430      return {
 431        reactions,
 432        points,
 433        sortedCrit,
 434        diagnostics: {
 435          maxV: Number(maxV.toFixed(3)),
 436          minV: Number(minV.toFixed(3)),
 437          maxAbsV: Number(maxAbsV.toFixed(3)),
 438          maxVPos: Number(maxVPos.toFixed(3)),
 439          maxM: Number(maxM.toFixed(3)),
 440          maxMPos: Number(maxMPos.toFixed(3)),
 441          minM: Number(minM.toFixed(3)),
 442          minMPos: Number(minMPos.toFixed(3)),
 443          maxAbsM: Number(maxAbsM.toFixed(3)),
 444          peakStress: Number(peakStress.toFixed(2)),
 445          stressRatio: Number(stressRatio.toFixed(1)),
 446          safetyFactor: Number(isFinite(safetyFactor) ? safetyFactor.toFixed(2) : '99.9'),
 447          zeroShearPoints,
 448          contraflexurePoints,
 449          totalLoad: Number(reactions.totalDownwardForce.toFixed(3)),
 450          totalReaction: Number(reactions.totalUpwardReaction.toFixed(3)),
 451          balanceCheck: reactions.balanceError < 1e-4
 452        }
 453      };
 454    }

 456    /**
 457     * Find all points where Shear Force V(x) = 0
 458     * These correspond to local extrema of Bending Moment M(x)
 459     * Captures both continuous slope crossings inside UDLs and step jumps across zero at point loads/reactions
 460     */
 461    findZeroShearPoints(points, boundaries, reactions) {
 462      const results = [];
 463      if (this.pointLoads.length === 0 && this.udlLoads.length === 0 && this.moments.length === 0) {
 464        return results;
 465      }
 466  
 467      const eps = 1e-5;
 468  
 469      // 1. Check for step jumps across zero at boundary points (point loads & reactions)
 470      for (const bx of boundaries) {
 471        if (bx <= 0.01 || bx >= this.length - 0.01) continue;
 472  
 473        const evalBefore = this.evaluateAt(bx - eps, reactions);
 474        const evalAfter = this.evaluateAt(bx + eps, reactions);
 475  
 476        if (evalBefore.V * evalAfter.V <= 0 && Math.abs(evalAfter.V - evalBefore.V) > 0.05) {
 477          const rootEval = this.evaluateAt(bx, reactions);
 478          if (!results.some(r => Math.abs(r.x - bx) < 0.05)) {
 479            results.push({
 480              x: Number(bx.toFixed(3)),
 481              V: 0,
 482              M: Number(Math.max(Math.abs(evalBefore.M), Math.abs(evalAfter.M)) * (evalBefore.M >= 0 ? 1 : -1)).toFixed(3),
 483              type: 'Zero Shear Step / Peak Moment'
 484            });
 485          }
 486        }
 487      }
 488  
 489      // 2. Check for continuous zero crossings inside spans between boundaries
 490      for (let i = 0; i < boundaries.length - 1; i++) {
 491        const xA = boundaries[i];
 492        const xB = boundaries[i + 1];
 493        if (xB - xA < 1e-4) continue;
 494  
 495        const evalA = this.evaluateAt(xA + eps, reactions);
 496        const evalB = this.evaluateAt(xB - eps, reactions);
 497  
 498        if (evalA.V * evalB.V < -1e-5) {
 499          const dV = evalB.V - evalA.V;
 500          let rootX;
 501          if (Math.abs(dV) > 1e-6) {
 502            rootX = xA + (-evalA.V / dV) * (xB - xA);
 503          } else {
 504            rootX = (xA + xB) / 2;
 505          }
 506  
 507          rootX = Math.max(xA, Math.min(xB, rootX));
 508          const rootEval = this.evaluateAt(rootX, reactions);
 509  
 510          if (!results.some(r => Math.abs(r.x - rootX) < 0.05)) {
 511            results.push({
 512              x: Number(rootX.toFixed(3)),
 513              V: 0,
 514              M: Number(rootEval.M.toFixed(3)),
 515              type: 'Zero Shear / Peak Moment'
 516            });
 517          }
 518        }
 519      }
 520  
 521      results.sort((a, b) => a.x - b.x);
 522      return results;
 523    }

 525    /**
 526     * Find Points of Contraflexure (where Bending Moment M(x) = 0)
 527     * Inflection points between sagging and hogging zones (curvature changes sign)
 528     */
 529    findContraflexurePoints(points, boundaries, reactions) {
 530      const results = [];
 531      if (this.pointLoads.length === 0 && this.udlLoads.length === 0 && this.moments.length === 0) {
 532        return results;
 533      }
 534  
 535      for (let i = 0; i < points.length - 1; i++) {
 536        const p1 = points[i];
 537        const p2 = points[i + 1];
 538  
 539        // Exclude boundary zeros at free beam ends
 540        if (p1.x < 0.05) continue;
 541        if (p2.x > this.length - 0.05) continue;
 542  
 543        // Strict sign crossing: p1.M * p2.M < 0
 544        if (p1.M * p2.M < -1e-6) {
 545          let rootX = p1.x;
 546          const dM = p2.M - p1.M;
 547          if (Math.abs(dM) > 1e-6) {
 548            rootX = p1.x + (-p1.M / dM) * (p2.x - p1.x);
 549          }
 550          rootX = Number(rootX.toFixed(3));
 551  
 552          if (rootX > 0.05 && rootX < this.length - 0.05 && !results.some(r => Math.abs(r.x - rootX) < 0.08)) {
